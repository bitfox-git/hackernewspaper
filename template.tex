% \fbfont{class}{font}: enter the font when a run starts (after a normal character
% or a boundary), leave it when the run ends
\newcommand\fbfont[2]{%
  \XeTeXinterchartoks 0 #1 = {\begingroup#2}%
  \XeTeXinterchartoks 4095 #1 = {\begingroup#2}%
  \XeTeXinterchartoks #1 0 = {\endgroup}%
  \XeTeXinterchartoks #1 4095 = {\endgroup}}
% \fbswitch{from class}{to class}{font}: one fallback run directly followed by another
\newcommand\fbswitch[3]{\XeTeXinterchartoks #1 #2 = {\endgroup\begingroup#3}}
